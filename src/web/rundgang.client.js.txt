/*
 * Der Rundgang — die einzige Seite, die ohne Netz funktioniert.
 *
 * Kein Framework, kein Bundler: dieses Skript wird als Text einkompiliert und im Seitenkopf
 * ausgeliefert. Es hält drei Dinge in IndexedDB — die Daten der Begehung, eine Warteschlange
 * von Operationen und die noch nicht hochgeladenen Fotos — und schiebt sie hoch, sobald Netz
 * da ist. Was einmal getippt ist, ist sicher; das Netz entscheidet nur, wann es ankommt.
 */
(function () {
  "use strict";

  var BEG = document.body.dataset.begehung;
  var DB_NAME = "tuerwerk";
  var db = null;
  var daten = null;
  var ops = [];
  var fotos = [];
  var offen = null; // gerade geöffnetes Bauteil

  /* ── IndexedDB, so klein wie möglich ──────────────────────────────────── */

  /*
   * Die Prüfpunkte kommen vom Türtyp, nicht von der Vorlage: umbenannt, ausgeblendet, um eigene
   * ergänzt. Nur wenn eine Tür keinen Typ hat, bleibt die rohe Vorlage übrig.
   */
  function vorlageFuer(b) {
    var v = daten.vorlagen[b.art] || daten.vorlagen[Object.keys(daten.vorlagen)[0]];
    var c = daten.checklisten && b.tuertyp_id ? daten.checklisten[b.tuertyp_id] : null;
    if (!c || !c.punkte || !c.punkte.length) return v;
    return {
      label: c.name || v.label,
      punkte: c.punkte,
      bauteilfelder: v.bauteilfelder,
      labels: v.labels,
      kurz: v.kurz,
    };
  }

  function dbOeffnen() {
    return new Promise(function (fertig, fehler) {
      var a = indexedDB.open(DB_NAME, 1);
      a.onupgradeneeded = function () {
        var d = a.result;
        if (!d.objectStoreNames.contains("daten")) d.createObjectStore("daten");
        if (!d.objectStoreNames.contains("ops")) d.createObjectStore("ops", { keyPath: "op_id" });
        if (!d.objectStoreNames.contains("fotos")) d.createObjectStore("fotos", { keyPath: "op_id" });
      };
      a.onsuccess = function () { fertig(a.result); };
      a.onerror = function () { fehler(a.error); };
    });
  }

  function tx(laden, art, arbeit) {
    return new Promise(function (fertig, fehler) {
      var t = db.transaction(laden, art);
      var ergebnis;
      t.oncomplete = function () { fertig(ergebnis); };
      t.onerror = function () { fehler(t.error); };
      ergebnis = arbeit(t);
    });
  }

  function alleLesen(laden) {
    return tx([laden], "readonly", function (t) {
      var a = t.objectStore(laden).getAll();
      return new Promise(function (f) { a.onsuccess = function () { f(a.result || []); }; });
    }).then(function (p) { return p; });
  }

  function setzen(laden, wert, schluessel) {
    return tx([laden], "readwrite", function (t) {
      schluessel === undefined ? t.objectStore(laden).put(wert) : t.objectStore(laden).put(wert, schluessel);
    });
  }

  function entfernen(laden, schluessel) {
    return tx([laden], "readwrite", function (t) { t.objectStore(laden).delete(schluessel); });
  }

  function holen(laden, schluessel) {
    return tx([laden], "readonly", function (t) {
      var a = t.objectStore(laden).get(schluessel);
      return new Promise(function (f) { a.onsuccess = function () { f(a.result); }; });
    }).then(function (p) { return p; });
  }

  /* ── Kleinkram ────────────────────────────────────────────────────────── */

  function uuid() {
    if (crypto.randomUUID) return crypto.randomUUID();
    return "op-" + Date.now() + "-" + Math.random().toString(36).slice(2, 10);
  }

  function esc(s) {
    return String(s == null ? "" : s).replace(/[&<>"']/g, function (z) {
      return { "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;" }[z];
    });
  }

  function ort(b) {
    return [b.geschoss, b.raumnummer, b.raum || b.bezeichnung, b.flur].filter(Boolean).join(" · ")
      || "ohne Ortsangabe";
  }

  function el(id) { return document.getElementById(id); }

  /* ── Zustand: Server-Daten plus das, was noch in der Warteschlange steht ─ */

  function ortMerkmale(b) {
    var p = b.pruefung;
    if (!p) return '<span class="marke">' + b.nr + "</span>";
    var n = Object.keys(p.checks || {}).length;
    return '<span class="marke">' + (n ? n : "✓") + "</span>";
  }

  function zustandAnwenden() {
    /* Wartende Operationen über die Serverdaten legen, damit die Liste sofort stimmt. */
    ops.forEach(function (op) {
      if (op.art === "bauteil_neu") {
        if (!daten.bauteile.some(function (b) { return b.nr === op.payload.nr; })) {
          daten.bauteile.push({
            id: "lokal-" + op.op_id, nr: op.payload.nr, kennung: "", art: op.payload.art,
            raumnummer: op.payload.raumnummer || "", raum: op.payload.raum || "", flur: "",
            bezeichnung: "", geschoss: "", felder: {}, letzte: null, maengel: [], pruefung: null,
            lokal: true,
          });
        }
      }
      if (op.art === "pruefung") {
        var b = daten.bauteile.filter(function (x) { return x.nr === op.payload.nr; })[0];
        if (b) {
          b.pruefung = {
            checks: op.payload.checks || {}, ergebnis: op.payload.ergebnis || "bestanden",
            hinweise: op.payload.hinweise || "", wartet: true,
          };
          if (op.payload.felder) Object.assign(b.felder, op.payload.felder);
          if (op.payload.raum) b.raum = op.payload.raum;
          if (op.payload.raumnummer) b.raumnummer = op.payload.raumnummer;
          if (op.payload.flur) b.flur = op.payload.flur;
        }
      }
      if (op.art === "mangel_schliessen") {
        var t = daten.bauteile.filter(function (x) { return x.nr === op.payload.nr; })[0];
        if (t) t.maengel = [];
      }
    });
  }

  function naechsteNr() {
    var hoechste = 0;
    daten.bauteile.forEach(function (b) { if (b.nr > hoechste) hoechste = b.nr; });
    return hoechste + 1;
  }

  /* ── Anzeige ──────────────────────────────────────────────────────────── */

  function standZeigen() {
    var wartend = ops.length + fotos.length;
    var s = el("stand");
    s.textContent = wartend
      ? wartend + (wartend === 1 ? " Änderung wartet" : " Änderungen warten")
      : "alles übertragen";
    s.className = "abgleich" + (wartend ? " wartet" : "");
    if (!navigator.onLine) { s.textContent += " · kein Netz"; }
  }

  function listeZeigen() {
    offen = null;
    el("detail").hidden = true;
    el("liste").hidden = false;
    var geprueft = daten.bauteile.filter(function (b) { return b.pruefung; }).length;
    var gesamt = daten.bauteile.length;
    el("zahl").textContent = geprueft + " von " + gesamt;
    el("fuellung").style.width = (gesamt ? Math.round((geprueft / gesamt) * 100) : 0) + "%";

    var naechstes = daten.bauteile.filter(function (b) { return !b.pruefung; })[0];
    el("dran").innerHTML = naechstes
      ? '<div class="was">JETZT DRAN</div><div class="wer">Tür ' + naechstes.nr +
        '</div><div class="wo">' + esc(ort(naechstes)) + "</div>"
      : '<div class="was">FERTIG</div><div class="wer">Alle Türen erfasst</div>' +
        '<div class="wo">Zurück ins Gespräch: Claude liest zurück.</div>';
    if (naechstes) el("dran").dataset.nr = naechstes.nr; else delete el("dran").dataset.nr;

    el("haken").innerHTML = daten.bauteile.map(function (b) {
      var p = b.pruefung;
      var n = p ? Object.keys(p.checks || {}).length : 0;
      var klasse = "reihe" + (p ? " fertig" : "") + (n ? " abweichung" : "");
      var unten = p
        ? (p.ergebnis + (n ? " · Punkt " + Object.keys(p.checks).join(", ") : "") +
           (p.wartet ? " · wartet" : ""))
        : ort(b);
      var mangel = b.maengel && b.maengel.length
        ? '<span class="chip mangel">' + b.maengel.length + " offen</span>" : "";
      return '<button class="' + klasse + '" data-nr="' + b.nr + '">' + ortMerkmale(b) +
        '<span class="txt">Tür ' + b.nr + "<small>" + esc(unten) + "</small></span>" + mangel + "</button>";
    }).join("");
    standZeigen();
  }

  function detailZeigen(nr) {
    var b = daten.bauteile.filter(function (x) { return x.nr === nr; })[0];
    if (!b) return;
    offen = b;
    var v = vorlageFuer(b);
    var p = b.pruefung || { checks: {}, ergebnis: "bestanden", hinweise: "" };

    /* Wo steht die Tür? Bei einer im Rundgang angelegten weiß das sonst niemand. */
    var ortsfelder =
      '<div class="feld"><label for="o_raumnummer">Raumnummer</label>' +
      '<input class="field" id="o_raumnummer" value="' + esc(b.raumnummer || "") + '"></div>' +
      '<div class="feld"><label for="o_raum">Raum</label>' +
      '<input class="field" id="o_raum" value="' + esc(b.raum || "") + '"></div>' +
      '<div class="feld"><label for="o_flur">Flur</label>' +
      '<input class="field" id="o_flur" value="' + esc(b.flur || "") + '"></div>';
    var felder = ortsfelder + v.bauteilfelder.map(function (f) {
      return '<div class="feld"><label for="f_' + f + '">' + esc(v.labels[f] || f) + "</label>" +
        '<input class="field" id="f_' + f + '" data-feld="' + f + '" value="' + esc(b.felder[f] || "") + '"></div>';
    }).join("");

    var punkte = v.punkte.map(function (pt) {
      var gewaehlt = p.checks[pt.nr] || "io";
      var knoepfe = ["io", "nio", "nz", "sb"].map(function (bw) {
        return '<label><input type="radio" name="p_' + pt.nr + '" value="' + bw + '"' +
          (gewaehlt === bw ? " checked" : "") + "><span>" + v.kurz[bw] + "</span></label>";
      }).join("");
      return '<div class="punkt"><div class="txt"><b>' + pt.nr + "</b>" + esc(pt.text) +
        '</div><div class="wahl">' + knoepfe + "</div></div>";
    }).join("");

    var maengel = (b.maengel || []).length
      ? '<div class="note" style="margin:18px 0">Offen seit früher: ' +
        b.maengel.map(function (m) { return esc(m.beschreibung || "Mangel"); }).join(" · ") +
        ' <button class="btn schmal leise" id="freimelden" type="button">Behoben</button></div>'
      : "";

    var letzte = b.letzte
      ? '<p class="meta">Zuletzt ' + esc(b.letzte.datum) + " · " + esc(b.letzte.ergebnis) + "</p>"
      : '<p class="meta">Noch nie geprüft.</p>';

    el("detail").innerHTML =
      '<button class="btn schmal leise" id="zurueck" type="button">‹ Liste</button>' +
      '<h1 class="seite" style="margin-top:16px">Tür ' + b.nr + "</h1>" +
      '<p class="meta" style="margin-top:6px">' + esc(ort(b)) + "</p>" + letzte + maengel +
      '<div class="felder" style="margin-top:18px">' + felder +
      '<div class="feld"><label for="ergebnis">Ergebnis</label><select class="field" id="ergebnis">' +
      '<option value="bestanden"' + (p.ergebnis === "bestanden" ? " selected" : "") + ">bestanden</option>" +
      '<option value="Nachbesserung"' + (p.ergebnis === "Nachbesserung" ? " selected" : "") + ">Nachbesserung</option>" +
      "</select></div></div>" +
      '<h2 class="abschnitt">Prüfpunkte</h2><div class="punkte">' + punkte + "</div>" +
      '<div class="feld" style="margin-top:20px"><label for="hinweise">Hinweise</label>' +
      '<textarea class="field" id="hinweise">' + esc(p.hinweise) + "</textarea></div>" +
      '<div class="knopfleiste"><button class="btn schmal" id="sichern" type="button">Tür speichern</button>' +
      '<label class="btn schmal leise" for="kamera">Foto</label>' +
      '<input id="kamera" type="file" accept="image/*" capture="environment" hidden>' +
      '<span class="meta" id="fotostand"></span></div>';

    el("liste").hidden = true;
    el("detail").hidden = false;
    window.scrollTo(0, 0);
  }

  /* ── Speichern ────────────────────────────────────────────────────────── */

  function pruefungSichern() {
    if (!offen) return;
    var b = offen;
    var v = vorlageFuer(b);
    var felder = {};
    v.bauteilfelder.forEach(function (f) {
      var eingabe = document.querySelector('[data-feld="' + f + '"]');
      if (eingabe) felder[f] = eingabe.value.trim();
    });
    var checks = {};
    v.punkte.forEach(function (pt) {
      var gewaehlt = document.querySelector('input[name="p_' + pt.nr + '"]:checked');
      if (gewaehlt && gewaehlt.value !== "io") checks[pt.nr] = gewaehlt.value;
    });
    var op = {
      op_id: uuid(),
      art: "pruefung",
      ts: Date.now(),
      payload: {
        nr: b.nr,
        art: b.art,
        checks: checks,
        ergebnis: el("ergebnis").value,
        hinweise: el("hinweise").value.trim(),
        felder: felder,
        raumnummer: el("o_raumnummer").value.trim(),
        raum: el("o_raum").value.trim(),
        flur: el("o_flur").value.trim(),
        geprueft_am: Date.now(),
      },
    };
    ops.push(op);
    setzen("ops", op);
    b.pruefung = { checks: checks, ergebnis: op.payload.ergebnis, hinweise: op.payload.hinweise, wartet: true };
    Object.assign(b.felder, felder);
    b.raumnummer = op.payload.raumnummer;
    b.raum = op.payload.raum;
    b.flur = op.payload.flur;
    listeZeigen();
    abgleichen();
  }

  function neueTuer() {
    var nr = naechsteNr();
    /* Türtyp und Vorlage der zuletzt erfassten Tür — im Treppenhaus ist das fast immer richtig. */
    var letzte = daten.bauteile[daten.bauteile.length - 1] || {};
    var art = letzte.art || Object.keys(daten.vorlagen)[0];
    var typId = letzte.tuertyp_id || "";
    var op = {
      op_id: uuid(), art: "bauteil_neu", ts: Date.now(),
      payload: { nr: nr, art: art, tuertyp_id: typId },
    };
    ops.push(op);
    setzen("ops", op);
    daten.bauteile.push({
      id: "lokal-" + op.op_id, nr: nr, kennung: "", art: art, tuertyp_id: typId,
      raumnummer: "", raum: "", flur: "",
      bezeichnung: "", geschoss: "", felder: {}, letzte: null, maengel: [], pruefung: null, lokal: true,
    });
    speichern();
    detailZeigen(nr);
    abgleichen();
  }

  function freimelden() {
    if (!offen) return;
    var op = {
      op_id: uuid(), art: "mangel_schliessen", ts: Date.now(),
      payload: { nr: offen.nr, freimeldung: "Im Rundgang als behoben gemeldet" },
    };
    ops.push(op);
    setzen("ops", op);
    offen.maengel = [];
    detailZeigen(offen.nr);
    abgleichen();
  }

  /* ── Foto: im Browser verkleinern, dann in die Warteschlange ──────────── */

  function fotoAufnehmen(datei) {
    if (!datei || !offen) return;
    var nr = offen.nr;
    el("fotostand").textContent = "Foto wird verkleinert …";
    createImageBitmap(datei, { imageOrientation: "from-image" }).then(function (bild) {
      var faktor = Math.min(1600 / Math.max(bild.width, bild.height), 1);
      var leinwand = document.createElement("canvas");
      leinwand.width = Math.round(bild.width * faktor);
      leinwand.height = Math.round(bild.height * faktor);
      var stift = leinwand.getContext("2d");
      /* Weiß unterlegen: ein durchsichtiges PNG würde als JPEG sonst schwarz. */
      stift.fillStyle = "#fff";
      stift.fillRect(0, 0, leinwand.width, leinwand.height);
      stift.drawImage(bild, 0, 0, leinwand.width, leinwand.height);
      return new Promise(function (f) {
        leinwand.toBlob(function (blob) {
          f({ blob: blob, breite: leinwand.width, hoehe: leinwand.height });
        }, "image/jpeg", 0.82);
      });
    }).then(function (r) {
      var eintrag = {
        op_id: uuid(), nr: nr, blob: r.blob, breite: r.breite, hoehe: r.hoehe, ts: Date.now(),
      };
      fotos.push(eintrag);
      return setzen("fotos", eintrag);
    }).then(function () {
      el("fotostand").textContent = "Foto gemerkt";
      standZeigen();
      abgleichen();
    }).catch(function () {
      el("fotostand").textContent = "Foto ging nicht";
    });
  }

  /* ── Abgleich ─────────────────────────────────────────────────────────── */

  var laeuft = false;

  function abgleichen() {
    if (laeuft || !navigator.onLine) { standZeigen(); return Promise.resolve(); }
    if (!ops.length && !fotos.length) { standZeigen(); return Promise.resolve(); }
    laeuft = true;
    var zuSenden = ops.slice();
    return fetch("/api/sync", {
      method: "POST",
      headers: { "content-type": "application/json" },
      body: JSON.stringify({ begehung: BEG, ops: zuSenden }),
    }).then(function (r) {
      if (!r.ok) throw new Error("sync " + r.status);
      return r.json();
    }).then(function (antwort) {
      var umgelegt = {};
      (antwort.ergebnisse || []).forEach(function (e) {
        if (e.zuordnung) umgelegt[e.zuordnung.client_nr] = e.zuordnung.server_nr;
        if (e.ok) {
          ops = ops.filter(function (o) { return o.op_id !== e.op_id; });
          entfernen("ops", e.op_id);
        }
      });
      /* Nummern, die der Server umgelegt hat, in wartenden Operationen und Fotos nachziehen. */
      Object.keys(umgelegt).forEach(function (alt) {
        ops.forEach(function (o) {
          if (String(o.payload.nr) === alt) { o.payload.nr = umgelegt[alt]; setzen("ops", o); }
        });
        fotos.forEach(function (f) {
          if (String(f.nr) === alt) { f.nr = umgelegt[alt]; setzen("fotos", f); }
        });
      });
      return fotosSenden();
    }).then(function () {
      return neuLaden();
    }).catch(function () {
      /* Kein Netz oder Serverfehler: alles bleibt liegen, der nächste Lauf holt es. */
    }).then(function () {
      laeuft = false;
      standZeigen();
    });
  }

  function fotosSenden() {
    if (!fotos.length) return Promise.resolve();
    var naechstes = fotos[0];
    var form = new FormData();
    form.append("op_id", naechstes.op_id);
    form.append("begehung_id", BEG);
    form.append("bauteil_nr", String(naechstes.nr));
    form.append("breite", String(naechstes.breite));
    form.append("hoehe", String(naechstes.hoehe));
    form.append("bild", naechstes.blob, "foto.jpg");
    return fetch("/api/foto", { method: "POST", body: form }).then(function (r) {
      if (!r.ok) throw new Error("foto " + r.status);
      fotos = fotos.filter(function (f) { return f.op_id !== naechstes.op_id; });
      return entfernen("fotos", naechstes.op_id);
    }).then(function () {
      return fotosSenden();
    });
  }

  function neuLaden() {
    return fetch("/api/rundgang/" + BEG, { headers: { accept: "application/json" } })
      .then(function (r) { return r.ok ? r.json() : null; })
      .then(function (frisch) {
        if (!frisch) return;
        daten = frisch;
        zustandAnwenden();
        speichern();
        if (!offen) listeZeigen(); else standZeigen();
      });
  }

  function speichern() {
    return setzen("daten", daten, BEG);
  }

  /* ── Start ────────────────────────────────────────────────────────────── */

  document.addEventListener("click", function (e) {
    var reihe = e.target.closest(".reihe[data-nr]");
    if (reihe) { detailZeigen(Number(reihe.dataset.nr)); return; }
    if (e.target.closest("#dran") && el("dran").dataset.nr) {
      detailZeigen(Number(el("dran").dataset.nr)); return;
    }
    if (e.target.id === "zurueck") { listeZeigen(); return; }
    if (e.target.id === "sichern") { pruefungSichern(); return; }
    if (e.target.id === "freimelden") { freimelden(); return; }
    if (e.target.id === "neu") { neueTuer(); return; }
    if (e.target.id === "stand" || e.target.id === "jetzt") { abgleichen(); return; }
  });

  document.addEventListener("change", function (e) {
    if (e.target.id === "kamera") fotoAufnehmen(e.target.files[0]);
  });

  window.addEventListener("online", abgleichen);
  window.addEventListener("offline", standZeigen);
  setInterval(abgleichen, 30000);

  dbOeffnen().then(function (d) {
    db = d;
    return Promise.all([alleLesen("ops"), alleLesen("fotos"), holen("daten", BEG)]);
  }).then(function (r) {
    ops = r[0] || [];
    fotos = r[1] || [];
    var zwischengespeichert = r[2];
    return fetch("/api/rundgang/" + BEG, { headers: { accept: "application/json" } })
      .then(function (a) { return a.ok ? a.json() : null; })
      .catch(function () { return null; })
      .then(function (frisch) {
        daten = frisch || zwischengespeichert;
        if (!daten) {
          el("liste").innerHTML =
            '<div class="leer">Diese Begehung war noch nie offen und es ist kein Netz da.<br>' +
            "Einmal mit Netz öffnen, danach geht sie auch ohne.</div>";
          return;
        }
        if (frisch) speichern();
        zustandAnwenden();
        listeZeigen();
        abgleichen();
      });
  });

  if ("serviceWorker" in navigator) {
    navigator.serviceWorker.register("/sw.js").catch(function () {});
  }
})();
