/*
 * Service Worker des Rundgangs.
 *
 * Nur zwei Regeln: die Seite selbst und ihre Beiwerke kommen aus dem Cache, sobald sie einmal
 * da waren (damit der Rundgang im Keller ohne Netz aufgeht); alles unter /api/ geht immer ans
 * Netz (damit niemand veraltete Daten für frisch hält). Der Cache-Name trägt die Version —
 * ein neues Deployment wirft den alten weg.
 */
var CACHE = "tuerwerk-__VERSION__";

self.addEventListener("install", function (e) {
  self.skipWaiting();
  e.waitUntil(caches.open(CACHE).then(function (c) { return c.addAll(["/icon.svg"]); }).catch(function () {}));
});

self.addEventListener("activate", function (e) {
  e.waitUntil(
    caches.keys().then(function (namen) {
      return Promise.all(namen.map(function (n) { return n === CACHE ? null : caches.delete(n); }));
    }).then(function () { return self.clients.claim(); }),
  );
});

self.addEventListener("fetch", function (e) {
  var url = new URL(e.request.url);
  if (e.request.method !== "GET" || url.origin !== self.location.origin) return;
  if (url.pathname.indexOf("/api/") === 0) return; // immer ans Netz

  var istRundgang = url.pathname.indexOf("/rundgang/") === 0;
  if (!istRundgang && url.pathname !== "/icon.svg") return;

  e.respondWith(
    caches.match(e.request).then(function (getroffen) {
      var ausDemNetz = fetch(e.request).then(function (antwort) {
        if (antwort && antwort.status === 200) {
          var kopie = antwort.clone();
          caches.open(CACHE).then(function (c) { c.put(e.request, kopie); });
        }
        return antwort;
      }).catch(function () { return getroffen; });
      /* Aus dem Cache sofort, im Hintergrund auffrischen — offline zählt jede Sekunde. */
      return getroffen || ausDemNetz;
    }),
  );
});
